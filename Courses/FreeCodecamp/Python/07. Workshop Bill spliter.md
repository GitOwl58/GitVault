# 07. Workshop bill splitter


``` python
running_total = 0
num_of_friends = 4
appetizers = 37.89
main_courses = 57.34
desserts = 39.39
drinks = 64.21

running_total += appetizers + main_courses + desserts + drinks

print('Total bill so far:', running_total) # Total bill so far: xxx.xxxxx

  
tip = running_total * 0.25

print('Tip amount:', tip) # Tip amount: xxx.xxxxx


running_total += tip

print('Total with tip:', running_total) # Total with tip: xxx.xxxxx

  
final_bill = running_total / num_of_friends

print('Bill per person:', final_bill) # Bill per person: xxx.xxxxx


each_pays = round(final_bill,2)

print(f'Each person pays:', each_pays) # Each person pays: xxx.xx
```
